### 성공 테스트 CLI 명령어


python - <<'PY'
from src.harness.safety import validate_sql

sql = """
SELECT
    Genre.Name,
    COUNT(Track.TrackId) AS track_count
FROM Genre
JOIN Track
    ON Genre.GenreId = Track.GenreId
GROUP BY Genre.Name
ORDER BY track_count DESC
LIMIT 5
"""

result = validate_sql(sql)

print("safe:", result.is_safe)
print("reason:", result.reason)
print("normalized:", result.normalized_sql)
PY

    ArtistId,
    Name
